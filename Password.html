import string import secrets def generate_password(length): characters = string.ascii_letters + string.digits + string.punctuation password = ''.join( secrets.choice(characters) for _ in range(length) ) return password print("===== PASSWORD GENERATOR =====") while True: try: length = int(input("Enter password length: ")) if length <= 0: print("Length must be greater than 0.") continue password = generate_password(length) print("\nGenerated Password:") print(password) again = input("\nGenerate another password? (yes/no): ").lower() if again != "yes": print("Password generator closed.") break except ValueError: print("Please enter a valid number.")
